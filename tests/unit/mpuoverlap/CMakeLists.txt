# Which overlaps of one thread's regions each region backend decides as the kernel's range checks
# do, through the kernel's predicate and the rule each backend's own mpu_encoded.h states. One
# translation unit per backend header, each of which shares one include guard.
set(_mo_rules "")
foreach(_mo armv6m_pmsav7 armv7m_pmsav7 armv7m_pmsav8 armv7m_sysmpu rv32imac rxv3 sim)
  list(APPEND _mo_rules "${CMAKE_CURRENT_BINARY_DIR}/rule_${_mo}.cc")
endforeach()
set(_mo_headers
  armv6m_pmsav7 "arch/arm/armv6m" KICKOS_ARM_MPU_PMSAV7
  armv7m_pmsav7 "arch/arm/armv7m" KICKOS_ARM_MPU_PMSAV7
  armv7m_pmsav8 "arch/arm/armv7m" KICKOS_ARM_MPU_PMSAV8
  armv7m_sysmpu "arch/arm/armv7m" KICKOS_ARM_MPU_SYSMPU
  rv32imac "arch/riscv/rv32imac" KICKOS_ARM_MPU_NONE
  rxv3 "arch/rx/rxv3" KICKOS_ARM_MPU_NONE
  sim "arch/sim" KICKOS_ARM_MPU_NONE)
while(_mo_headers)
  list(POP_FRONT _mo_headers _mo_name _mo_dir _mo_unit)
  set(_mo_header "${PROJECT_SOURCE_DIR}/${_mo_dir}/include/kickos/arch/mpu_encoded.h")
  if(NOT EXISTS "${_mo_header}")
    message(FATAL_ERROR "KickOS: ${_mo_header} is gone, so mpu_overlap would pin no rule for "
                        "${_mo_name}.")
  endif()
  file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/rule_${_mo_name}.cc" CONTENT
"// Generated by tests/unit/mpuoverlap/CMakeLists.txt.
#undef KICKOS_ARM_MPU
#define KICKOS_ARM_MPU ${_mo_unit}
#include \"${_mo_header}\"
int mpu_overlap_rule_${_mo_name}()
{
    return ARCH_MPU_OVERLAP;
}
")
endwhile()

kickos_add_unit_test(NAME mpu_overlap SOURCES mpu_overlap.cc ${_mo_rules} KERNEL)
