# Declarations for tests/static/check_trap_redzone.sh. This file, not the script and not
# the tool, says WHICH symbols a trap prologue can reach, WHICH macro holds the figure it
# enforces, and WHAT the graph is allowed not to know. check_trap_redzone.sh reads the
# macro names from here and resolves them through the compiler, under the flags the image was
# built with, so a figure lives in exactly one place: the header the assembler and the C both
# read. One of them is posture-dependent, which is why the compiler resolves it rather than
# a scrape of the source text.
#
# WHAT A CLASS IS. A trap prologue reserves a red zone at the bottom of the interrupted
# unprivileged stack and refuses the trap when less room than that remains below sp. The
# red zone is FRAME plus DEPTH: DEPTH is the half this gate MEASURES, the worst-case bytes
# the kernel C dispatch descends on that same stack, and FRAME is everything else in the
# zone. On rv32imac and rxv3 that remainder is exactly the frame the prologue stores; on
# armv7m it also carries the hardware frame a preempting exception stacks on the caller's
# PSP, which no call graph can see. The two halves stay separate macros: folded into one, a
# growing dispatch could eat the structural allowance while the gate still reported room to
# spare.
#
# A class is one (FRAME, DEPTH) pair with its own root set, and the hardware is what
# separates the entries: mcause on RISC-V, a separate vector on RX, a separate exception on
# ARM. Collapsing the classes would reserve the syscall figure under every timer tick.
#
# FILE FORMAT, one record per line, '#' starts a comment:
#   arch    <arch> header=<path-relative-to-source-dir>
#   preset  <arch> <configure-preset>
#   class   <arch> <CLASS> frame=<MACRO> depth=<MACRO> [stack=trap|stack=kernel] \
#                                                        [kstacks=0|kstacks=1]
#   root    <arch> <CLASS> [?]<symbol>
#   root    <arch> <CLASS> NONE reason: <text to end of line>
#   exclude <arch> <mangled-symbol> reason: <text to end of line>
#   unsized <arch> <symbol> <bytes> reason: <text to end of line>
#
# A '?' before a root marks it OPTIONAL: the symbol is compiled out under some option, so
# its absence from the graph contributes 0 instead of failing. A root with no '?' that is
# missing is a hard failure, because a root the graph does not have is a path nobody
# measured.
#
# '?' WORKS ON AN `exclude` TOO, and for the same reason `unsized` tolerates an absent
# allowance: these lists span the fleet, and a given image need not reach every symbol in
# them. Without it, excluding a symbol only SOME boards' graphs contain fails every board
# that inlines it away. A '?' exclusion the graph does not have is reported as unused.
#
# NONE declares that the class runs NO C on the guarded stack, so its measured depth is 0
# and there is nothing to walk. It is the deliberate twin of an omitted root: a class with
# no record at all is refused, because a root set that is empty by accident measures 0 and
# passes forever. It carries a reason because the claim is structural rather than measured.
#
# A `root` symbol is matched against the graph as an exact name or as the mangled tail of
# a file-scoped one; the match must be UNIQUE, and an ambiguous name is refused rather than
# guessed at.
#
# WHICH PRESETS BELONG HERE, AND WHY THE HONEST ANSWER IS "EVERY ONE OF THEM". A `preset`
# record is what makes a (board, variant) pair measurable at all, and the ctest ladder in the
# top-level CMakeLists then picks which of those run automatically. Both lists have been short
# three times, and each time a pair nobody measured was the defect: f302nucleo-st and
# bluepill-c8-st were never measured while the armv7m thread-stack class they depend on was
# live; after three boards converted, no REGISTERED armv7m preset measured that class at all;
# and the two rv32imac KICKOS_BENCH variants fell through the ladder's (board, variant) match
# in silence while their SYSPRIV depth stood 112 bytes over the figure it claimed.
#
# SO THE RULE IS A PROPERTY OF THE KNOBS AND NOT A LIST. A variant is MEASUREMENT-RELEVANT
# when it changes any of the following, and every one of them was MEASURED to move a class
# depth on a board in this tree rather than assumed to:
#
#   KICKOS_BENCH           compiles a syscall arm that prints, so a dispatch reaches the
#                          console from inside the bench bracket. rv32imac SYSPRIV 688 -> 816
#                          (esp32c6-wroom-st -> -bench); armv7m SVC 556 -> 760
#                          (xmc4800-relax -> -bench).
#   KICKOS_TELEMETRY       compiles the trace emitters and the ring drain arch_shutdown runs,
#                          which the dispatch reaches through exit_current and through the
#                          panic tail. armv7m SVCK 672 -> 1240 (qemu -> qemu-telem), the
#                          largest single posture effect in the fleet.
#   KICKOS_ENABLE_SELFTEST compiles syscall arms of its own, so it DOES add dispatch depth:
#                          rv32imac SYSPRIV 592 -> 688 and armv7m SVCK 672 -> 760
#                          (esp32c6-wroom -> -st, frdmk64f -> frdmk64f-st).
#   the memory model       MEMORY_MODEL_MPU against FLAT moves the grant and commit chains
#                          BOTH ways, so a flat variant is not automatically the cheaper one:
#                          armv7m SVC 556 -> 436 shallower (blackpill -> blackpill-flat), but
#                          rv32imac SYS 816 -> 832 DEEPER (esp32c6-wroom -> -flat).
#   the console backend    per BOARD rather than per variant, and it is the tail every
#                          printing chain ends in.
#
# A board's variants differ by exactly those knobs, so every preset of every arch below is
# measurement-relevant and belongs in both lists. WHAT THAT COSTS is one scratch cross-build
# per preset: about 45 MB and a cold minute each, reused across runs, and charged to whoever
# runs that board's ctest rather than to every board's. All four arches now carry their full
# set and every one of the 48 presets is registered: rv32imac 7, rxv3 3, armv6m 4, armv7m 34.
#
# AND THE LADDER IS NO LONGER A SECOND LIST. The top-level CMakeLists used to repeat these
# pairs as an if/elseif chain, which is how it came to be short three times: a missing arm
# registered nothing and a registered set is indistinguishable from a complete one. It now
# derives the preset name from the board and variant it is configuring and asks THIS FILE
# whether that name is declared, so a new preset enters both lists by being declared once.
#
# ARMV7M IS THE LAST TO ARRIVE, all 34, and closing it took two changes rather than one
# raise. All 34 were MEASURED BEFORE ANY FIGURE MOVED, which is what turned three known
# failures into the two distinct problems they actually were:
#
#   THE SVCK FAILURES WERE A CEILING TOO LOW, and the ceiling is posture-dependent.
#   xmc4800-relax-st and xmc4800-relax-bench measure 768 where the block provided 988 usable
#   against a 992 requirement, and qemu-telem measures 1240 through arch_shutdown's telemetry
#   tail. KICKOS_KERNEL_STACK_SIZE now resolves per KICKOS_TELEMETRY, 1008 and 1472, because
#   the blocks are per thread SLOT and a fleet-wide raise to the telemetry figure fails to
#   link bluepill-c8, which has no telemetry variant at all. Its slack was MEASURED rather
#   than argued, by bisecting the ceiling against the link: 1216 links and 1232 does not, so
#   224 per slot over 3 slots is 672 bytes, where the telemetry figure wants 480 per slot.
#   1008 spends 48 of that 672 and leaves 624. The refusal is KICKOS_POOL_ARENA_ASSERT in
#   arch/common/boot_arena.ld.h, the blocks being .bss below the arena rather than a term in
#   it, so they move __kickos_ram_start and the arena shrinks from underneath.
#
#   THE SVC FAILURES WERE A CLASS ENFORCED WHERE IT DOES NOT EXIST. xmc4800-relax-bench
#   measures 760 and qemu-telem 920 against 656, and neither board compiles the design that
#   class describes: after PR 5 the SVC depth guards f302nucleo and due alone, and neither of
#   those has a bench or a telemetry variant. Raising 656 to cover them would charge those two
#   boards for postures they never compile. So SVC carries kstacks=0 and the gate skips it,
#   loudly, wherever KICKOS_KERNEL_STACKS is 1.
#
# WHAT STILL ENFORCES THE SVC FIGURE, now that 30 of the 34 skip it, is f302nucleo,
# f302nucleo-st, due and due-st. That is the reason the full set matters rather than an
# argument against it: with only the four presets registered before this, the arch's two
# designs were covered by three presets and one of them measured the wrong one.
#
# WHICH ARCHES ARE HERE, AND WHY LX6 IS NOT. The four below are every backend that writes
# through a pointer an unprivileged thread chose, or once did, while an MPU is enforcing.
# lx6 is absent on purpose: CHIP_ESP32 selects no HAS_MPU, so nothing on that part isolates
# one thread's memory from another's and a wild-sp privileged descent reaches nothing a
# plain store from the thread itself does not already reach. A red zone there would cost
# bytes and buy no confinement. The day an lx6 part gains an enforcing backend, this file
# is where the omission has to be answered.
#
# stack=trap says this class's descent does NOT run on an unprivileged thread stack: it runs
# on a static kernel stack the arch owns. Two consequences follow from that one fact. The
# SPAWN FLOOR clause does not apply, because no thread's stack is being spent. And the
# declared `exclude` set does NOT apply either: a panic tail is excluded only because
# counting it would put a THREAD's red zone above the spawn floor, and a kernel array has no
# floor to clear.
#
# kstacks=<n> says this class belongs to ONE of the two entry designs an arch can compile,
# the one KICKOS_KERNEL_STACKS selects at <n>. Where the knob disagrees the image does not
# contain the path the class describes, so check_trap_redzone.sh measures it, PRINTS that it
# is not enforcing it, and drops both halves: the depth would be the same C measured landing
# on a stack this design does not put it on, and the red zone would charge the board for a
# dispatch it never runs there. Only armv7m needs it, being the only arch here that compiles
# two designs; rv32imac, rxv3 and armv6m each static_assert KICKOS_KERNEL_STACKS != 0, so
# every class of theirs is unconditional and carries no marker.
#
# IT IS NOT THE SAME FIELD AS stack=, and the two are not interchangeable. stack= says which
# stack a descent lands on, which is what picks the fit clause; kstacks= says whether the
# image contains that descent at all. The mirror of this marker is implemented off
# stack=kernel, a kernel-block class having neither a block to fit nor that descent to
# measure where the knob is 0, and that direction stays keyed there rather than restated as
# kstacks=1 on five records. It drops BOTH halves, as this marker does. It dropped only the
# fit half until 2026-08-22, so the gate printed that it was not enforcing SVCK and then
# enforced its depth; f302nucleo-st and due-st are the presets that showed it.
#
# stack=kernel says the same two things for the same reason, and adds a clause of its own:
# the descent runs on the interrupted thread's PER-THREAD KERNEL stack, one block of
# KICKOS_KERNEL_STACK_SIZE per thread slot in kernel .bss, so frame plus depth is a
# requirement on THAT knob and check_trap_redzone.sh compares it against the block. The
# lowest word of a block is its overflow canary, so the fit has to leave that word alone.
# stack=trap and stack=kernel are exclusive: a frame goes on one stack.

# --- rv32imac ------------------------------------------------------------------
# SEVEN classes. SIX ARE TRAP CLASSES and TWO registers separate them, both readable before the
# frame is built.
# mcause picks the entry: the .Lintr arms take TRAP, msip takes SWITCH, an ecall takes SYS or
# SYSPRIV, and any other exception takes FAULT. mstatus.MPP then picks the stack: an ecall from
# U-mode is SYS on the thread's kernel block, the same ecall from M-mode is SYSPRIV on the sp
# it interrupted, and every M-mode cause that is not msip or an ecall is NESTED on the kernel's
# own trap stack rather than on the stack it interrupted.
#
# THE SEVENTH IS NOT A TRAP CLASS AT ALL. EXIT measures the death path, which runs AFTER the
# entry has handed control back: the reporter accepts the fault, .Lfault relocates sp to the top
# of the dying thread's own stack and mrets into the exit stub, and the whole exit descent
# happens there, privileged, on the stack of a thread of any privilege. It is in this file
# because the SPAWN FLOOR clause is the only thing that bounds it and that clause lives here.
#
# TRAP AND SYS SPEND NO UNPRIVILEGED THREAD'S STACK. trap_entry's U-mode accept path loads
# ctx.kernel_sp and builds the frame on the interrupted thread's own kernel stack, so both
# classes are a requirement on KICKOS_KERNEL_STACK_SIZE rather than on the spawn floor:
# stack=kernel, which is also what makes the panic tail counted for them.
#
# SYSPRIV IS THAT SAME ECALL FROM A PRIVILEGED THREAD, which arrives with mstatus.MPP=M and so
# keeps its frame on the sp it interrupted, that thread's own stack, with svc_trampoline running
# the same dispatch there. Same FRAME_SYS, same roots, THREAD stack, and therefore the depth
# macro with the panic tail excluded: it is the one rv32imac trap class the SPAWN FLOOR clause
# still has anything to say about, and together with EXIT it is what sets
# KICKOS_MIN_STACK_SIZE on this arch.
#
# ITS DEPTH MACRO IS THE ONE POSTURE-DEPENDENT FIGURE IN THIS FILE, resolving per KICKOS_BENCH
# in rv_trap_stack.h because the bench syscall arm prints and nothing else compiles it. That is
# why check_trap_redzone.sh resolves the arch header through the compiler instead of reading it
# as text: one macro name, one live value, and the same number in the C floor assert.
#
# FAULT is the accepted U-mode fault: .Lfault runs the reporter with sp on the frame the entry
# built, which is now the kernel stack, so the whole dump chain lands there. Its depth is the
# same reporter chain KICKOS_RV_TRAP_NESTED_DEPTH already measures on the trap stack, and it is
# a class of its own because it is the KERNEL BLOCK it has to fit.
#
# SWITCH IS THE ONE CLASS STILL ON A THREAD STACK, and that is the reason it must not be
# deleted: .Lswitch and arch_start call kickos_arch_mpu_commit with sp = incoming->sp, and for
# a thread that has never trapped that is still the frame arch_context_init fabricated on its
# USER stack. Measuring 0 there is what says no privileged C frame stands on user memory.
# rv_trap_stack.h also composes this class INTO the syscall requirement, a blocking dispatch
# (or a tick's wake) taking the msip trap at whatever depth the dispatch had reached.
arch    rv32imac header=arch/riscv/rv32imac/include/kickos/arch/rv_trap_stack.h
preset  rv32imac qemu-riscv
preset  rv32imac qemu-riscv-flat
preset  rv32imac qemu-riscv-bench
preset  rv32imac esp32c6-wroom
preset  rv32imac esp32c6-wroom-st
preset  rv32imac esp32c6-wroom-flat
preset  rv32imac esp32c6-wroom-bench
class   rv32imac TRAP   frame=KICKOS_RV_TRAP_FRAME depth=KICKOS_RV_TRAP_KERNEL_DEPTH stack=kernel
class   rv32imac SYS    frame=KICKOS_RV_TRAP_FRAME_SYS depth=KICKOS_RV_TRAP_KERNEL_DEPTH_SYS stack=kernel
class   rv32imac SYSPRIV frame=KICKOS_RV_TRAP_FRAME_SYS depth=KICKOS_RV_TRAP_KERNEL_DEPTH_SYS_NO_PANIC
class   rv32imac FAULT  frame=KICKOS_RV_TRAP_FRAME depth=KICKOS_RV_TRAP_NESTED_DEPTH stack=kernel
class   rv32imac SWITCH frame=KICKOS_RV_TRAP_FRAME depth=KICKOS_RV_TRAP_SWITCH_DEPTH
class   rv32imac NESTED frame=KICKOS_RV_TRAP_FRAME depth=KICKOS_RV_TRAP_NESTED_DEPTH stack=trap
class   rv32imac EXITK  frame=KICKOS_RV_TRAP_NEST_EXIT depth=KICKOS_RV_TRAP_KERNEL_DEPTH_EXITK stack=kernel
class   rv32imac RET    frame=KICKOS_RV_TRAP_NEST_EXIT depth=KICKOS_RV_TRAP_KERNEL_DEPTH_RET

# The symbols the .Lintr arms of switch.S branch to.
root    rv32imac TRAP kickos_isr_timer
root    rv32imac TRAP kickos_rv_dispatch_soft
root    rv32imac TRAP kickos_rv_ext_dispatch
root    rv32imac TRAP kickos_rv_ext_dispatch_dev
# Called from the prologue/epilogue itself, on the same stack, so they are charged to
# every class. The telemetry and bench hooks do not exist unless their option is on.
root    rv32imac TRAP kickos_arch_mpu_commit
root    rv32imac TRAP ?kickos_trace_switch_done
root    rv32imac TRAP ?kickos_bench_switch_done
# The .Lintr demux calls this ahead of the arm it picked, on the same stack, in selftest
# builds only. A sibling of the ISR call rather than an ancestor, so it adds to this class's
# MAX and not to its winning chain.
root    rv32imac TRAP ?kickos_rv_nested_witness

# .Lecall_slow mrets into svc_trampoline, which calls syscall_dispatch; .Lecall calls
# kickos_ipc_fastpath inline, and the taken fastpath runs the same hooks.
root    rv32imac SYS  syscall_dispatch
root    rv32imac SYS  kickos_ipc_fastpath
root    rv32imac SYS  kickos_arch_mpu_commit
root    rv32imac SYS  ?kickos_trace_switch_done
root    rv32imac SYS  ?kickos_bench_switch_done

# SYSPRIV is the SAME entry reached from M-mode, so the root set is the SYS one exactly; what
# differs is the stack it lands on and, with it, whether the panic tail is charged.
root    rv32imac SYSPRIV syscall_dispatch
root    rv32imac SYSPRIV kickos_ipc_fastpath
root    rv32imac SYSPRIV kickos_arch_mpu_commit
root    rv32imac SYSPRIV ?kickos_trace_switch_done
root    rv32imac SYSPRIV ?kickos_bench_switch_done

# .Lfault, for a U-mode fault the entry ACCEPTED: the reporter and the stack-top query run on
# the frame the entry built, which is the thread's kernel block. The kill path returns early
# and the panic path walks the console; this charges the deeper of the two, so no claim about
# C control flow is needed.
root    rv32imac FAULT kickos_rv_fault_report
root    rv32imac FAULT kickos_fault_stack_top

# THE DEATH PATH, TWO CLASSES, because two of its three roots relocated and one could not. No
# trap prologue bounds either. EXITK is the fault and slay stubs on the thread's own KERNEL
# BLOCK: .Lfault relocates sp to kickos_fault_stack_top, which now answers with ctx.kernel_sp,
# and arch_ctx_redirect rebuilds the slay frame at that same top. RET is kickos_thread_return,
# a PRIVILEGED thread's entry-return stub, which nothing relocates, so the SPAWN FLOOR clause
# is the whole of ITS enforcement. The frame term is the msip frame on both: the stubs
# reschedule, and an msip trap from M-mode keeps its frame on the interrupted sp.
root    rv32imac EXITK kickos_thread_fault_exit
root    rv32imac EXITK kickos_thread_slay_exit
root    rv32imac RET   kickos_thread_return

# What .Lswitch itself calls, below the frame it just saved.
root    rv32imac SWITCH kickos_arch_mpu_commit
root    rv32imac SWITCH ?kickos_trace_switch_done
root    rv32imac SWITCH ?kickos_bench_switch_done

# The nested path: the four .Lintr arms again, because an M-mode entry runs them on the trap
# stack instead, plus .Lfault's reporter, which lands there for every M-mode fault and
# already did for the wild-sp refusal. kickos_arch_mpu_commit is NOT here: .Lswitch and
# .Lecall are the only callers and neither is a nested path.
root    rv32imac NESTED kickos_isr_timer
root    rv32imac NESTED kickos_rv_dispatch_soft
root    rv32imac NESTED kickos_rv_ext_dispatch
root    rv32imac NESTED kickos_rv_ext_dispatch_dev
root    rv32imac NESTED kickos_rv_fault_report
root    rv32imac NESTED kickos_fault_stack_top
root    rv32imac NESTED ?kickos_rv_nested_witness

# THE TWO EXCLUSIONS SERVE SYSPRIV, that being the rv32imac class whose figure they move. They
# do NOT reach TRAP, SYS or FAULT: a stack=kernel class is measured with nothing excluded, which
# is where the decision that a kernel array has no floor to clear is spent. SWITCH is on a
# thread stack too, but its roots cannot reach the tail, and if one ever did, failing it would be
# the right answer: that would be a privileged panic descent on user memory. EXIT is on a thread
# stack and the exclusions DO apply to it, and change nothing: it measures 624 and 720 with them
# and without them, the fault reporter's own console chain being deeper than the tail.
#
# ONE OF THE TWO IS ALSO WHAT MAKES THE SYSPRIV FIGURE POSTURE-DEPENDENT RATHER THAN CONSTANT.
# At KICKOS_BENCH 1 the bench arm's console chain outruns the tail, so the excluded and
# unexcluded figures coincide at 816 and the exclusion buys nothing there. It is kept because
# the non-bench posture still needs it.
exclude rv32imac _ZN6kickos6kpanicEPKc reason: the noreturn panic tail, which every KICKOS_ASSERT in the dispatch reaches. Counting it puts the SYSPRIV requirement at 1168, above every rv32imac KICKOS_MIN_STACK_SIZE, so a PRIVILEGED thread spawned at the floor could not make a syscall at all. The residual is stated in rv_trap_stack.h; the tool prints the without-exclusions figure so it cannot go quiet.
exclude rv32imac ?_ZN6kickos9kpanic_atEPKcj reason: the SECOND door into the same noreturn panic tail, which KICKOS_ASSERT reaches where kpanic takes a message. Excluding one and not the other left the tail counted on any board whose deepest chain happens to enter through this one, which is a figure the exclusion above was meant to remove. Same residual, stated once in rv_trap_stack.h.

# Assembly and libgcc bodies have no .ci entry, so their frames are declared here. None of
# the three references sp anywhere in its body and none calls out, so all three are
# frameless register-only leaves; riscv gcc reports 0 bytes (static) for a frameless leaf
# too, so 0 is the graph's own unit and no return-address slot is added. Re-derive with
# riscv32-none-elf-objdump -d --disassemble=<sym>.
unsized rv32imac __udivdi3 0 reason: measured from the ELF, frameless register-only leaf
unsized rv32imac __umoddi3 0 reason: measured from the ELF, frameless register-only leaf
unsized rv32imac __clzsi2  0 reason: measured from the ELF, frameless register-only leaf
unsized rv32imac __divdi3  0 reason: measured from libgcc.a for this multilib, frameless register-only leaf: its body touches sp nowhere and ends in a bare ret, exactly as the two above. It is in every board's graph and only a KICKOS_BENCH measurement REACHES it, that arm's print dividing signed 64-bit where nothing else in the trap path does.

# NOT AN UNMEASURED BODY BUT AN ALIAS. gcc emits the complete-object destructor as a triangle
# node with no "bytes (static)" and an edge to the base-object one, and the two are the same
# address: rx/riscv nm reports both BenchScope::~BenchScope() at offset 0 of one weak section.
# So 0 is its own frame and the edge below it carries the real cost, which the graph sizes at
# 0 as well. KICKOS_BENCH only, ktime_rearm being the caller.
unsized rv32imac _ZN6kickos10BenchScopeD1Ev 0 reason: measured from the object, a same-address alias of the base-object destructor rather than a body of its own

# --- rxv3 ----------------------------------------------------------------------
# FOUR classes. PENDSW is the deferred-switch vector (INTB[27]); the syscall trap then splits,
# because kickos_rx_syscall_trap's `cmp #56, r1` picks between a generic arm that stashes the
# [userPC][userPSW] head and RTEs into svc_trampoline, and a fastpath arm that saves the whole
# 236-byte context and runs kickos_ipc_fastpath inline. Different frame AND different descent,
# so each is its own class: folded together, KICKOS_RX_TRAP_KERNEL_DEPTH_SYS_FAST would be
# enforced by the assembly and measured by nobody.
#
# THE GENERIC ARM'S DISPATCH IS SYSK, stack=kernel. svc_trampoline's first act is to move R0
# onto the calling thread's own kernel block (ctx.kernel_sp), so syscall_dispatch and the tree
# under it are measured against KICKOS_KERNEL_STACK_SIZE and not against the spawn floor.
#
# THE SYS CLASS IS NOT DELETABLE FOR THAT, and it is the one class here with no root at all.
# The head the trap commits before its RTE still lands on the thread's own stack, because RTE
# pops PC then PSW from R0 and R0 is the USP while PSW.U is set, so the resume USP is that
# head's address plus eight. KICKOS_RX_TRAP_NEST_SYS prices the head plus the switcher zone a
# SWINT can put below it while svc_trampoline is still on that stack; no C frame is left there
# at all. rx_trap_stack.h derives every term.
arch    rxv3 header=arch/rx/rxv3/include/kickos/arch/rx_trap_stack.h
# Every rxv3 preset, per the knob rule above, and every one registered. rx72m-st is the deepest
# of the three (SYSK 788 against 692 base and 696 flat), so it is the one that sets the figure,
# and the other two are here because measuring the setter alone is what left the fleet unchecked
# three times over.
preset  rxv3 rx72m
preset  rxv3 rx72m-st
preset  rxv3 rx72m-flat
class   rxv3 PENDSW   frame=KICKOS_RX_TRAP_FRAME_PENDSW   depth=KICKOS_RX_TRAP_KERNEL_DEPTH_PENDSW
class   rxv3 SYS      frame=KICKOS_RX_TRAP_NEST_SYS       depth=KICKOS_RX_TRAP_KERNEL_DEPTH_SYS
class   rxv3 SYS_FAST frame=KICKOS_RX_TRAP_FRAME_SYS_FAST depth=KICKOS_RX_TRAP_KERNEL_DEPTH_SYS_FAST
class   rxv3 SYSK     frame=KICKOS_RX_TRAP_NEST_SYSK      depth=KICKOS_RX_TRAP_KERNEL_DEPTH_SYSK stack=kernel
# THE DEATH PATH. The backend rewrites the stacked PC so the RTE lands in a stub with R0 at
# kickos_fault_stack_top, which now answers with ctx.kernel_sp. EXITK is the two relocating
# stubs on the block; RET is kickos_thread_return, which cannot move.
# ARCH_KERNEL_STACKS_MANDATORY here, so no kstacks=0 fallback class is needed.
class   rxv3 EXITK    frame=KICKOS_RX_TRAP_NEST_EXIT      depth=KICKOS_RX_TRAP_KERNEL_DEPTH_EXITK stack=kernel
class   rxv3 RET      frame=KICKOS_RX_TRAP_NEST_EXIT      depth=KICKOS_RX_TRAP_KERNEL_DEPTH_RET

root    rxv3 PENDSW kickos_arch_mpu_commit
root    rxv3 PENDSW ?kickos_trace_switch_done
