# The kernel-text instructions that materialise an APP-HALF address and are allowed to
# (tests/static/check_riscv_kernel_apphalf.sh).
#
# One symbol per line, never a pattern, each with the reason the reference is harmless. A
# reference that DEREFERENCES an app-half address does not belong here whatever it is for: it
# reaches whichever process is on the core, not the one the kernel meant.
#
# An entry no kernel-text instruction reaches any more is stale and fails the gate.

kickos_root_entry           # a PC: kmain hands it to thread_create as the root thread's entry, never dereferenced
kickos_user_thread_return   # a PC: arch_context_init seats it as the return address of an unprivileged frame, never dereferenced
_kickos_heap_start          # a bound: kmain prints the newlib heap's size, never dereferenced
_kickos_heap_limit          # a bound: the other end of the same printed subtraction, never dereferenced
